} } // 大小王 this.deck.push(new Card('SJ', '', 'joker')); this.deck.push(new Card('BJ', '', 'joker')); } shuffle() { for (let i = this.deck.length - 1; i > 0; i--) { const j = Math.floor(Math.random() * (i + 1)); [this.deck[i], this.deck[j]] = [this.deck[j], this.deck[i]]; } } deal() { this.shuffle(); this.players.left = this.deck.slice(0, 17); this.players.right = this.deck.slice(17, 34); this.players.player = this.deck.slice(34, 51); this.holeCards = this.deck.slice(51, 54); // 排序手牌 this.sortHand(this.players.left); this.sortHand(this.players.right); this.sortHand(this.players.player); this.renderHands(); this.updateCounts(); } sortHand(hand) { hand.sort((a, b) => b.value - a.value); } // --- 渲染相关 --- renderHands() { const container = document.getElementById('player-hand-container'); container.innerHTML = ''; this.players.player.forEach((card, index) => { const el = this.createCardElement(card); 'left' : 'right'; this.showMessage(lucky, "抢地主"); this.setLandlord(lucky); } } setLandlord(who) { this.landlord = who; this.state = 'playing'; this.turn = who; // 地主先出 // 发放底牌 const landlordHand = this.players[who]; landlordHand.push(...this.holeCards); this.sortHand(landlordHand); // 更新UI显示底牌 const holeContainer = document.getElementById('hole-cards'); holeContainer.innerHTML = ''; this.holeCards.forEach(c => { const el = this.createCardElement(c); el.classList.remove('card'); // 移除交互样式 el.classList.add('w-10', 'h-14', 'text-xs'); holeContainer.appendChild(el); }); // 更新角色标识 document.getElementById('player-role').innerText = who === 'player' ? // 44+333 if (cards.value === cards.value && cards.value === cards.value) return { type: 'triplet_pair', value: cards[2].value, count: 5, cards: cards }; } // 顺子等复杂牌型暂略，保持代码简洁 return null; } compareHand(current, last) { if (current.type === 'rocket') return true; else this.endGame(true); return true; } return false; } endGame(playerWin) { this.state = 'ended'; document.getElementById('action-area').innerHTML = `
